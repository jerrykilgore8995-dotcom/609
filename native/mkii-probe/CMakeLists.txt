cmake_minimum_required(VERSION 3.22)
project(609LinkProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(GenerateSolve ../GenerateSolve.cpp)
target_include_directories(GenerateSolve PRIVATE "${DECK_DIR}")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/GeneratedSolve.hpp"
    COMMAND GenerateSolve "${CMAKE_CURRENT_BINARY_DIR}/GeneratedSolve.hpp"
    DEPENDS GenerateSolve)
add_custom_target(SolveSchedule DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/GeneratedSolve.hpp")
foreach(probe FastLinkProbe)
    add_executable(${probe} LinkProbe.cpp)
    add_dependencies(${probe} SolveSchedule)
    target_include_directories(${probe} PRIVATE .. "${DECK_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
endforeach()
target_compile_definitions(FastLinkProbe PRIVATE MKII_CODE_SPEED)
if(MSVC)
    target_compile_options(GenerateSolve PRIVATE /arch:AVX2)
    target_compile_options(FastLinkProbe PRIVATE /arch:AVX2 /fp:precise)
endif()
